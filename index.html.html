# Student Management API — student_api.py
# Built with Python Flask · Run: python student_api.py
# ─────────────────────────────────────────────────────────────────

from flask import Flask, jsonify, request
from flask_cors import CORS

app = Flask(__name__)
CORS(app)  # Allow cross-origin requests from any browser

# ── In-memory data store (replaces a database for this demo) ──────
students = [
    {"id": 1, "name": "Alice Mwamba",   "age": 20, "course": "Computer Science",    "grade": "A"},
    {"id": 2, "name": "Brian Chanda",   "age": 22, "course": "Information Systems", "grade": "B+"},
    {"id": 3, "name": "Chola Tembo",    "age": 21, "course": "Software Eng.",       "grade": "A-"},
    {"id": 4, "name": "Diana Phiri",    "age": 23, "course": "Data Science",        "grade": "B"},
    {"id": 5, "name": "Emmanuel Banda", "age": 20, "course": "Cybersecurity",       "grade": "A+"},
]
next_id = 6

def find_student(sid):
    return next((s for s in students if s["id"] == sid), None)


# ── GET /api/students → Fetch ALL students ────────────────────────
@app.route("/api/students", methods=["GET"])
def get_all_students():
    return jsonify({
        "status": "success",
        "count": len(students),
        "students": students
    }), 200


# ── GET /api/students/:id → Fetch ONE student ─────────────────────
@app.route("/api/students/<int:student_id>", methods=["GET"])
def get_student(student_id):
    student = find_student(student_id)
    if not student:
        return jsonify({
            "status": "error",
            "message": f"Student with ID {student_id} not found"
        }), 404
    return jsonify({"status": "success", "student": student}), 200


# ── POST /api/students → Create a new student ─────────────────────
@app.route("/api/students", methods=["POST"])
def create_student():
    global next_id
    data = request.get_json()
    if not data:
        return jsonify({"status": "error", "message": "Request body must be JSON"}), 400

    required = ["name", "age", "course", "grade"]
    missing = [f for f in required if f not in data]
    if missing:
        return jsonify({
            "status": "error",
            "message": f"Missing required fields: {', '.join(missing)}"
        }), 400

    new_student = {"id": next_id, "name": data["name"],
                   "age": data["age"], "course": data["course"], "grade": data["grade"]}
    students.append(new_student)
    next_id += 1
    return jsonify({"status": "success", "message": "Student created",
                    "student": new_student}), 201


# ── PUT /api/students/:id → Update a student ──────────────────────
@app.route("/api/students/<int:student_id>", methods=["PUT"])
def update_student(student_id):
    student = find_student(student_id)
    if not student:
        return jsonify({"status": "error",
                        "message": f"Student {student_id} not found"}), 404

    data = request.get_json()
    for field in ["name", "age", "course", "grade"]:
        if field in data:
            student[field] = data[field]
    return jsonify({"status": "success", "message": "Student updated",
                    "student": student}), 200


# ── DELETE /api/students/:id → Remove a student ───────────────────
@app.route("/api/students/<int:student_id>", methods=["DELETE"])
def delete_student(student_id):
    student = find_student(student_id)
    if not student:
        return jsonify({"status": "error",
                        "message": f"Student {student_id} not found"}), 404
    students.remove(student)
    return jsonify({"status": "success",
                    "message": f"Student {student_id} deleted successfully"}), 200


# ── GET /api/health → Health check ───────────────────────────────
@app.route("/api/health", methods=["GET"])
def health():
    return jsonify({"status": "ok", "message": "Student API is running"}), 200


if __name__ == "__main__":
    print("Starting on http://localhost:5000")
    app.run(debug=True, port=5000)
